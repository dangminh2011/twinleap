cmake_minimum_required(VERSION 3.21)
project(twinleap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
 set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(TWINLEAP_FETCH_SDL3 "Download SDL3" ON)
option(TWINLEAP_BUILD_TESTS "Build tests" ON)
option(TWINLEAP_WINDOWS_GUI "Run without a console window on Windows" ON)
find_package(SDL3 3.2 CONFIG QUIET)
if(NOT SDL3_FOUND)
 if(NOT TWINLEAP_FETCH_SDL3)
  message(FATAL_ERROR "Install SDL3 or enable TWINLEAP_FETCH_SDL3")
 endif()
 include(FetchContent)
 foreach(opt SDL_SHARED SDL_TEST_LIBRARY SDL_TESTS SDL_INSTALL)
  set(${opt} OFF CACHE BOOL "" FORCE)
 endforeach()
 set(SDL_STATIC ON CACHE BOOL "" FORCE)
 FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-3.2.0 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(SDL3)
endif()
add_library(twinleap_core STATIC src/core/App.cpp src/core/Input.cpp src/game/Level.cpp src/game/Physics.cpp src/game/Player.cpp src/game/Entities.cpp src/game/Camera.cpp src/game/Particles.cpp src/game/MenuState.cpp src/game/PlayState.cpp src/render/Renderer.cpp src/render/Assets.cpp src/audio/Sfx.cpp)
option(TWINLEAP_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(TWINLEAP_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT MSVC)
 target_compile_options(twinleap_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(twinleap_core PUBLIC -fsanitize=address,undefined)
endif()
target_include_directories(twinleap_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(TARGET SDL3::SDL3)
 target_link_libraries(twinleap_core PUBLIC SDL3::SDL3)
else()
 target_link_libraries(twinleap_core PUBLIC SDL3::SDL3-static)
endif()
if(MSVC)
 target_compile_options(twinleap_core PRIVATE /W4)
else()
 target_compile_options(twinleap_core PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wconversion)
endif()
add_executable(twinleap src/main.cpp)
if(WIN32 AND TWINLEAP_WINDOWS_GUI)
 set_target_properties(twinleap PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
if(TARGET SDL3::SDL3-shared)
 add_custom_command(TARGET twinleap POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:twinleap>)
endif()
target_link_libraries(twinleap PRIVATE twinleap_core)
add_custom_target(twinleap_resources ALL
 COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:twinleap>/assets
 COMMENT "Copying game assets and levels")
add_dependencies(twinleap twinleap_resources)
install(TARGETS twinleap RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(FILES README.md DESTINATION .)
if(WIN32)
 if(TARGET SDL3::SDL3-shared)
  install(FILES $<TARGET_FILE:SDL3::SDL3-shared> DESTINATION .)
 endif()
 set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
 include(InstallRequiredSystemLibraries)
 set(CPACK_GENERATOR ZIP)
 set(CPACK_PACKAGE_NAME Twinleap)
 set(CPACK_PACKAGE_VERSION 1.0.0)
 set(CPACK_PACKAGE_FILE_NAME Twinleap-Windows-x64)
 include(CPack)
endif()
if(TWINLEAP_BUILD_TESTS)
 enable_testing()
 add_subdirectory(tests)
endif()
